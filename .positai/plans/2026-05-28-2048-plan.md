# Plano: Criar `6_Genomic_CRE_Annotation.Rmd`

## O que o usuário quer
Executar o pipeline de anotação de CREs em um arquivo .Rmd separado, cobrindo:
1. Importação e estruturação (homólogos + não-homólogos)
2. Anotação de genes associados (ChIPseeker)
3. Extração de sequências (BSgenome)
4. Alinhamento e identidade (Biostrings, local, gap -2/-1)
5. PhastCons — 7 vertebrados (ou fallback 60 mamíferos)
6. Elementos Transponíveis (RepeatMasker via AnnotationHub)
7. Estatística descritiva e exportação

### FASE 1 — Importação e Estruturação

```r
# Import homologous CREs from ENCODE
homologous_cres <- read_tsv(
  here("Genomic", "hg38-mm10-Homologous.tsv"),
  col_names = c("human_enhancer", "mouse_enhancer"),
  show_col_types = FALSE
)

# Import human CREs
cres_human <- read_tsv(
  here("Genomic", "GRCh38-cCREs.bed"),
  col_names = c("chrom", "start", "end", "id_dhs", "id_ccre", "type"),
  show_col_types = FALSE
) %>%
  rename_with(~paste0(., "_human"))

# Import mouse CREs
cres_mouse <- read_tsv(
  here("Genomic", "mm10-cCREs.bed"),
  col_names = c("chrom", "start", "end", "id_dhs", "id_ccre", "type"),
  show_col_types = FALSE
) %>%
  rename_with(~paste0(., "_mouse"))

# Contar matches por ID para classificar match_type
human_counts <- homologous_cres %>% count(human_enhancer, name = "n_mouse_per_human")
mouse_counts  <- homologous_cres %>% count(mouse_enhancer, name = "n_human_per_mouse")

# Full join para preservar não-homólogos
cres_annotated <- homologous_cres %>%
  left_join(human_counts, by = "human_enhancer") %>%
  left_join(mouse_counts,  by = "mouse_enhancer") %>%
  full_join(cres_human, by = join_by(human_enhancer == id_ccre_human)) %>%
  full_join(cres_mouse, by = join_by(mouse_enhancer == id_ccre_mouse)) %>%
  mutate(
    is_homologous = !is.na(human_enhancer) & !is.na(mouse_enhancer),
    match_type = case_when(
      is.na(human_enhancer)              ~ "mouse_only",
      is.na(mouse_enhancer)              ~ "human_only",
      n_mouse_per_human == 1 & n_human_per_mouse == 1 ~ "1:1",
      n_mouse_per_human >  1 & n_human_per_mouse == 1 ~ "N:1",
      n_mouse_per_human == 1 & n_human_per_mouse >  1 ~ "1:N",
      TRUE                               ~ "N:M"
    )
  )

# Tabelas separadas
cres_homologous   <- cres_annotated %>% filter(is_homologous)
cres_human_only   <- cres_annotated %>% filter(match_type == "human_only")
cres_mouse_only   <- cres_annotated %>% filter(match_type == "mouse_only")
```

**Nota:** O `full_join` com os CREs humanos e murinos pelo `id_ccre` irá incluir NAs nas linhas de espécie oposta. Isso é intencional — permite manter os 3 grupos em uma tabela única.

```r
  seqnames = cres_human_valid$chrom_human,
  ranges   = IRanges(start = cres_human_valid$start_human, end = cres_human_valid$end_human)

seqlevels(cres_human_gr) <- seqlevels(BSgenome.Hsapiens.UCSC.hg38)
seqinfo(cres_human_gr)   <- seqinfo(BSgenome.Hsapiens.UCSC.hg38)

anno_human <- annotatePeak(
  cres_human_gr,
  tssRegion = c(-3000, 3000),
  TxDb      = txdb_human,
  annoDb    = "org.Hs.eg.db"
) %>% as.data.frame()

# Adicionar anotação na tabela principal (apenas para linhas com coords humanas)
cres_annotated <- cres_annotated %>%
  mutate(
    gene_name_human     = ifelse(!is.na(chrom_human), anno_human$SYMBOL[match(id_ccre_human, cres_human_valid$human_enhancer)], NA),
    gene_id_human       = ifelse(!is.na(chrom_human), anno_human$ENTREZID[match(id_ccre_human, cres_human_valid$human_enhancer)], NA),
    annotation_human    = ifelse(!is.na(chrom_human), anno_human$annotation, NA),
    dist_tss_human      = ifelse(!is.na(chrom_human), anno_human$distanceToTSS, NA)
  )
```

**Problema potencial:** O `annotatePeak()` retorna 1 gene por CRE (o mais próximo). Se quiser todos os genes num raio, pode usar `findOverlaps()` com `genes(txdb)` + uma janela customizada. Vamos começar com o mais próximo e expandir depois se necessário.

### FASE 4 — Alinhamento e Identidade

```r
library(Biostrings)

# Função helper para um par de sequências
align_pair <- function(seq1, seq2,
                       gap_open = -2, gap_ext = -1) {
  if (is.na(seq1) || is.na(seq2) || nchar(seq1) == 0 || nchar(seq2) == 0) {
    return(list(identity = NA_real_, score = NA_real_, length = NA_integer_))
  }
  aln <- pairwiseAlignment(
    DNAString(seq1), DNAString(seq2),
    type          = "local",
    gapOpening    = gap_open,
    gapExtension  = gap_ext,
    scoreOnly     = FALSE
  )
  list(
    identity = pid(aln, type = "PID1"),   # matches / aligned length
    score    = score(aln),
    length   = nchar(alignedSubject(aln))
  )
}

# Aplica apenas em CREs homólogos (os mais custosos)
# Se >100k pares, considerar processar em paralelo com furrr
cre_alignments <- cres_homologous %>%
  select(human_enhancer, mouse_enhancer, sequence_cre_human, sequence_cre_mouse) %>%
  pmap(function(human_enhancer, mouse_enhancer, sequence_cre_human, sequence_cre_mouse) {
    aln <- align_pair(sequence_cre_human, sequence_cre_mouse)
    tibble(
      human_enhancer    = human_enhancer,
      mouse_enhancer    = mouse_enhancer,
      cre_identity_pct  = aln$identity,
      cre_align_score   = aln$score,
      cre_align_length  = aln$length
    )
  }) %>%
  bind_rows()

# Para promotores (mesmo gene humano vs murino):
promoter_alignments <- cres_homologous %>%
  select(human_enhancer, mouse_enhancer, sequence_promoter_human, sequence_promoter_mouse) %>%
  pmap(function(human_enhancer, mouse_enhancer, sequence_promoter_human, sequence_promoter_mouse) {
    aln <- align_pair(sequence_promoter_human, sequence_promoter_mouse)
    tibble(
      human_enhancer         = human_enhancer,
      mouse_enhancer         = mouse_enhancer,
      promoter_identity_pct  = aln$identity,
      promoter_align_score   = aln$score
    )
  }) %>%
  bind_rows()

# Juntar de volta
cres_annotated <- cres_annotated %>%
  left_join(cre_alignments,       by = c("human_enhancer", "mouse_enhancer")) %>%
  left_join(promoter_alignments,  by = c("human_enhancer", "mouse_enhancer"))
```

**Nota de performance:** 537k pares de alinhamento em single thread pode levar ~1-2h. Podemos usar `future` + `furrr::future_pmap()` para paralelizar se ficar lento.

```r
ah <- AnnotationHub()

# Buscar phastCons 7 vertebrados para hg38
phast_query <- query(ah, c("phastCons", "hg38", "7way"))

# Ver opções disponíveis
print(phast_query)

# Usar o primeiro resultado disponível de 7-way; se não houver, cair para 60way
if (length(phast_query) > 0) {
  phast_bw <- phast_query[[1]]
} else {
  phast_query_60 <- query(ah, c("phastCons", "hg38", "60way"))
  phast_bw <- phast_query_60[[1]]
  message("Usando phastCons 60-way como fallback")
}

# Importar scores apenas nas regiões dos CREs humanos (evita carregar o genoma todo)
phast_scores <- import(phast_bw, which = cres_human_gr, as = "RleList")

# Calcular média por CRE
phastcons_mean <- binnedAverage(cres_human_gr, phast_scores, "phastcons_mean")

cres_annotated <- cres_annotated %>%
  mutate(phastcons_human = ifelse(
    !is.na(chrom_human),
    phastcons_mean$phastcons_mean,
    NA_real_
  ))
```

```r
# Overlap
hits_human <- findOverlaps(cres_human_gr, rmsk_hg38, ignore.strand = TRUE)

te_human <- tibble(
  cre_idx    = queryHits(hits_human),
  te_name    = rmsk_hg38$repName[subjectHits(hits_human)],
  te_class   = rmsk_hg38$repClass[subjectHits(hits_human)],
  te_family  = rmsk_hg38$repFamily[subjectHits(hits_human)]
)

# Colapsar: 1 linha por CRE (pegar classe mais frequente, flag booleano)
te_human_summary <- te_human %>%
  group_by(cre_idx) %>%
  summarise(
    te_overlap_human  = TRUE,
    te_class_human    = paste(unique(te_class), collapse = ";"),
    te_family_human   = paste(unique(te_family), collapse = ";"),
    te_n_elements     = n()
  )

# Cobrir índices sem overlap
te_human_full <- tibble(cre_idx = seq_along(cres_human_gr)) %>%
  left_join(te_human_summary, by = "cre_idx") %>%
  mutate(te_overlap_human = replace_na(te_overlap_human, FALSE))

  mutate(
    te_overlap_human = te_human_full$te_overlap_human,
    te_class_human   = te_human_full$te_class_human,
```

```r
# Sumário geral
cre_summary <- cres_annotated %>%
  count(match_type) %>%
  mutate(
    pct = round(100 * n / sum(n), 1)

# Identidade por tipo de match
identity_by_type <- cres_homologous %>%
  group_by(match_type) %>%
  summarise(
    n              = n(),
    mean_identity  = mean(cre_identity_pct, na.rm = TRUE),
    median_identity = median(cre_identity_pct, na.rm = TRUE),
    pct_high_cons  = mean(cre_identity_pct > 80, na.rm = TRUE) * 100

cre_type_summary <- cres_annotated %>%
  group_by(type_human, is_homologous) %>%
```

## Observações e Riscos

| Item | Risco | Mitigação |
|---|---|---|
| Alinhamento de 537k pares | Pode levar 1-3h | Paralelizar com `furrr`; ou salvar checkpoint por cromossomo |
| PhastCons 7-way hg38 | Pode não estar no AnnotationHub | Fallback automático para 60-way |
| RepeatMasker AnnotationHub ID | IDs mudam entre versões | Inspecionar `rmsk_query` antes de usar |
| `annotatePeak()` com 1M CREs | Memória alta (~8-16GB) | Rodar por cromossomo se necessário |
| Linhas duplicadas no join N:M | Explosão combinatória | Documentado em `match_type`; usar `cres_homologous` apenas para alinhamento |
